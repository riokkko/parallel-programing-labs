cmake_minimum_required(VERSION 4.3.0-rc2)

project("labs" CXX)

add_executable("l2")

set(CMAKE_CXX_MODULE_STD ON)
set(CMAKE_EXPERIMENTAL_CXX_IMPORT_STD d0edc3af-4c50-42ea-a356-e2862fe7a444)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenMP REQUIRED)

set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /utf-8 /W4")

if(MSVC)
     add_compile_options(/std:c++23)
endif()

target_link_libraries(l2 PRIVATE OpenMP::OpenMP_CXX)

target_compile_features("l2" PUBLIC cxx_std_23)

if(MSVC)
    target_compile_options(l2 PRIVATE /openmp)
else()
    target_compile_options(l2 PRIVATE -fopenmp)
endif()

target_sources("l2" PRIVATE "l2.cpp" 
               PRIVATE FILE_SET "modules" TYPE CXX_MODULES 
               FILES "maker.cppm")
